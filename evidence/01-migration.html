export function PrivateRoute({ component: C, ...rest }) {

  return (
    <Route {...rest} render={props => {
      if (status === 'loading') return null;
      if (!user) return (

          pathname: '/login',
          state: { referrer: props.location },
        }} />
      );

    }} />
  );
}
export function RequireAuth() {
  const { user } = useSession();

}
